> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Architecture:** 新增 `agent_kb_router.resolve_agent_route()` 在预检索后决定路径；`agent.py` 按路由分支到 `stream_kb_fast_path_events` 或 `stream_agent_chat`；hermes_lite 通过增强预检索摘要 + Hermes context JSON 限制重复 MCP。

**Tech Stack:** FastAPI, existing Hermes SSE client, React/Next.js Agent 页, pytest.

**Design spec:** [specs/features/1.2.3-agent-kb-router-design.md](../features/1.2.3-agent-kb-router-design.md)

```python
# backend/services/agent_kb_router.py
from enum import Enum
import re
from typing import Any, Literal

AgentMode = Literal["auto", "fast", "standard", "deep"]

class AgentRoute(str, Enum):
    fast = "fast"
    hermes_lite = "hermes_lite"
    hermes_full = "hermes_full"

_SIMPLE_PAT = re.compile(
    r"^(?:.{0,40}(?:多少|是什么|什么意思|如何理解).{0,20}|.{0,30}(?:营收|收入).{0,20})$",
    re.I,
)
_COMPLEX_PAT = re.compile(r"对比|比较|两年|核实|再查|验证|口径|母公司|合并", re.I)

def _is_simple_kb_query(query: str) -> bool:
    q = (query or "").strip()
    if not q or _COMPLEX_PAT.search(q):
        return False
    return bool(_SIMPLE_PAT.search(q)) or (len(q) <= 32 and "?" not in q and "？" not in q)

def resolve_agent_route(
    *,
    user_query: str,
    agent_mode: AgentMode,
    allow_kb_write: bool,
    has_kb_scope: bool,
    prefetch_result: dict[str, Any] | None,
    hermes_configured: bool,
) -> AgentRoute:
    from backend.services.agent_kb_fast_path import prefetch_has_usable_evidence, query_implies_kb_write
    from backend.config import settings

    if not has_kb_scope:
        return AgentRoute.hermes_full
    if allow_kb_write and query_implies_kb_write(user_query):
        return AgentRoute.hermes_full
    if agent_mode == "deep":
        return AgentRoute.hermes_full
    if agent_mode == "fast" and prefetch_has_usable_evidence(prefetch_result):
        return AgentRoute.fast
    if agent_mode == "fast":
        return AgentRoute.hermes_lite  # 证据不足时降级 lite
    if agent_mode in ("standard", "auto") and getattr(settings, "hermes_agent_simple_query_fast", True):
        if agent_mode == "auto" and _is_simple_kb_query(user_query) and prefetch_has_usable_evidence(prefetch_result):
            cites = (prefetch_result or {}).get("citations") or []
            if len(cites) >= 3:
                return AgentRoute.fast
    if hermes_configured:
        return AgentRoute.hermes_lite
    return AgentRoute.fast
```
